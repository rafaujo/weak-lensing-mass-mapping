#   - FAIR maps

#   - evaluation metric
#
# DWT:
#
#   R_DWT(x) =
#       sum_{j,o} alpha_{j,o} ||c_{j,o}||_1

#

#
#   R_star(x) =
#       sum_j alpha_j ||w_j||_1
#
#   Redundant transform => primal-dual solver.
#
# Screen:
#   cosmologies C0,C1,C2
#   realizations R0,R3

def find_local_file(filename):

    cwd = Path.cwd()

    candidates = [
        cwd / filename,
        cwd / "data" / filename,
        cwd / "DATA" / filename,
        cwd / "FAIR" / filename,
        cwd / "fair" / filename,
    ]

    for p in candidates:

        if p.exists():
            return p.resolve()

    matches = list(
        cwd.rglob(filename)
    )

    if len(matches) == 0:
        raise FileNotFoundError(
            filename
        )

    return matches[0].resolve()

def periodic_response_1d(
    N,
    taps,
    offsets,
    device,
):

    kernel = torch.zeros(
        N,
        dtype=torch.float32,
        device=device,
    )

    for weight, offset in zip(
        taps,
        offsets,
    ):

        kernel[
            int(offset) % N
        ] += float(
            weight
        )

    return torch.fft.fft(
        kernel
    ).to(
        torch.complex64
    )


class StarletFrame:

    def __init__(
        self,
        H,
        W,
        J,
        device,
    ):

        self.H = int(H)
        self.W = int(W)
        self.J = int(J)

        self.device = torch.device(
            device
        )

        h = (
            np.array(
                [
                    1.0,
                    4.0,
                    6.0,
                    4.0,
                    1.0,
                ],
                dtype=np.float64,
            )
            /
            16.0
        )

        offsets0 = np.array(
            [
                -2,
                -1,
                0,
                1,
                2,
            ],
            dtype=int,
        )

        cumulative = torch.ones(
            self.H,
            self.W,
            dtype=torch.complex64,
            device=self.device,
        )

        transfers = []

        for level in range(
            1,
            self.J + 1,
        ):

            dilation = (
                2 **
                (
                    level - 1
                )
            )

            offsets = (
                offsets0
                *
                dilation
            )

            Hy = periodic_response_1d(
                self.H,
                h,
                offsets,
                self.device,
            )

            Hx = periodic_response_1d(
                self.W,
                h,
                offsets,
                self.device,
            )

            smooth = (
                Hy[:, None]
                *
                Hx[None, :]
            )

            detail = (
                cumulative
                *
                (
                    1.0
                    -
                    smooth
                )
            )

            transfers.append(
                detail
            )

            cumulative = (
                cumulative
                *
                smooth
            )

        self.transfers = torch.stack(
            transfers,
            dim=0,
        )

        self.coarse_transfer = (
            cumulative
        )

        power = torch.sum(
            torch.abs(
                self.transfers
            ) ** 2,
            dim=0,
        )

        self.norm_sq = float(
            power.max().item()
        )


    def analysis(
        self,
        x,
    ):

        X = torch.fft.fft2(
            x
        )

        C = (
            self.transfers
            *
            X[
                None,
                :,
                :
            ]
        )

        return (
            torch.fft.ifft2(
                C,
                dim=(-2, -1),
            )
            .real
            .float()
        )


    def adjoint(
        self,
        coeff,
    ):

        C = torch.fft.fft2(
            coeff,
            dim=(-2, -1),
        )

        X = torch.sum(
            torch.conj(
                self.transfers
            )
            *
            C,
            dim=0,
        )

        return (
            torch.fft.ifft2(
                X
            )
            .real
            .float()
        )


    def adjoint_test(
        self,
        seed=1234,
    ):

        g = torch.Generator(
            device=self.device
        )

        g.manual_seed(
            seed
        )

        x = torch.randn(
            self.H,
            self.W,
            generator=g,
            device=self.device,
        )

        p = torch.randn(
            self.J,
            self.H,
            self.W,
            generator=g,
            device=self.device,
        )

        Kx = self.analysis(
            x
        )

        KTp = self.adjoint(
            p
        )

        lhs = torch.sum(
            Kx * p
        )

        rhs = torch.sum(
            x * KTp
        )

        denom = (
            Kx.norm()
            *
            p.norm()
            +
            1e-30
        )

        return float(
            (
                torch.abs(
                    lhs - rhs
                )
                /
                denom
            ).item()
        )

# ============================================================
# 6. PROPAGATED NOISE CALIBRATION
#
# Same eta for both transforms:
#
#   eta = A^T M n_gamma
#
# This makes the DWT and Starlet calibration directly
# comparable.
# ============================================================

print("\n" + "=" * 118)
print("PROPAGATED-NOISE CALIBRATION")
print("=" * 118)


# Starlet:
#
# index 0..J-1 = levels 1..J.
star_var = torch.zeros(
    J,
    dtype=torch.float64,
    device=DEVICE,
)


# DWT:
#
# PyWavelets detail ordering:
#
#   details[0] = level J
#   ...
#   details[J-1] = level 1
#
# axis 1:
#   0=H, 1=V, 2=D
dwt_var = np.zeros(
    (
        J,
        3,
    ),
    dtype=np.float64,
)


for mc in range(
    N_NOISE_CAL
):

    g = torch.Generator(
        device=DEVICE
    )

    g.manual_seed(
        NOISE_CAL_SEED0
        +
        mc
    )


    n1 = (
        SIGMA_N
        *
        torch.randn(
            H,
            W,
            generator=g,
            device=DEVICE,
        )
    )

    n2 = (
        SIGMA_N
        *
        torch.randn(
            H,
            W,
            generator=g,
            device=DEVICE,
        )
    )


    # M n
    noise_gamma = torch.complex(
        mask_t
        *
        n1,

        mask_t
        *
        n2,
    )


    # eta = A^T M n
    eta = op.adjoint(
        noise_gamma
    )


    # --------------------------------------------------------
    # Starlet
    # --------------------------------------------------------

    sc = starlet.analysis(
        eta
    )

    star_var += torch.mean(
        sc.double() ** 2,
        dim=(-2, -1),
    )


    # --------------------------------------------------------
    # DWT
    # --------------------------------------------------------

    dc = base_dwt.decompose(
        eta
    )


    for i, (
        cH,
        cV,
        cD,
    ) in enumerate(
        dc["details"]
    ):

        dwt_var[
            i,
            0,
        ] += float(
            torch.mean(
                cH.double() ** 2
            ).item()
        )

        dwt_var[
            i,
            1,
        ] += float(
            torch.mean(
                cV.double() ** 2
            ).item()
        )

        dwt_var[
            i,
            2,
        ] += float(
            torch.mean(
                cD.double() ** 2
            ).item()
        )


# ============================================================

star_sigma = torch.sqrt(
